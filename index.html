freMTPL2 Feature Engineering Pipeline

Build a modelling-ready frequency table from the raw freMTPL2 data:

  • Actuarial age bands for driver and vehicle age
  • Log-exposure offset for Poisson / Negative-Binomial GLM
  • BonusMalus capping and log-density transform
  • Stratified 80/20 train/test split balanced on claim frequency

CREATE OR REPLACE TABLE FREMTPL2.FREMTPL2_INVESTIGATION.FREQ_MODEL_FEATURES AS
WITH base AS (
  SELECT
    IDPOL,
    CLAIMNB,
    EXPOSURE,
    VEHPOWER,
    VEHAGE,
    DRIVAGE,
    BONUSMALUS,
    VEHBRAND,
    VEHGAS,
    AREA,
    DENSITY,
    REGION,

    -- Log-exposure offset (Poisson/NB GLM offset term)
    LN(EXPOSURE)          AS LN_EXPOSURE,

    -- Driver age bands (actuarial risk curve)
    CASE
      WHEN DRIVAGE BETWEEN 18 AND 21 THEN '18-21'
      WHEN DRIVAGE BETWEEN 22 AND 25 THEN '22-25'
      WHEN DRIVAGE BETWEEN 26 AND 30 THEN '26-30'
      WHEN DRIVAGE BETWEEN 31 AND 40 THEN '31-40'
      WHEN DRIVAGE BETWEEN 41 AND 50 THEN '41-50'
      WHEN DRIVAGE BETWEEN 51 AND 60 THEN '51-60'
      WHEN DRIVAGE BETWEEN 61 AND 70 THEN '61-70'
      ELSE '71+'
    END                   AS DRIVAGE_BAND,

    -- Vehicle age bands (mechanical risk curve)
    CASE
      WHEN VEHAGE BETWEEN 0  AND 1  THEN '00-01'
      WHEN VEHAGE BETWEEN 2  AND 5  THEN '02-05'
      WHEN VEHAGE BETWEEN 6  AND 10 THEN '06-10'
      WHEN VEHAGE BETWEEN 11 AND 15 THEN '11-15'
      ELSE '16+'
    END                   AS VEHAGE_BAND,

    -- BonusMalus capped at 150 (outlier compression)
    LEAST(BONUSMALUS, 150) AS BONUSMALUS_CAPPED,

    -- Log density (de-skew for GLM)
    LN(DENSITY + 1)       AS LN_DENSITY,

    -- Binary claim flag for stratified splitting
    CASE WHEN CLAIMNB > 0 THEN 1 ELSE 0 END AS HAS_CLAIM

  FROM FREMTPL2.FREMTPL2_INVESTIGATION.FREMTPL2FREQ
),

stratified AS (
  SELECT
    b.*,
    ROW_NUMBER() OVER (
      PARTITION BY HAS_CLAIM
      ORDER BY ABS(HASH(IDPOL))
    ) AS RN,
    COUNT(*) OVER (PARTITION BY HAS_CLAIM) AS STRATUM_SIZE
  FROM base b
)

SELECT
  IDPOL,
  CLAIMNB,
  EXPOSURE,
  LN_EXPOSURE,
  VEHPOWER,
  VEHAGE,
  VEHAGE_BAND,
  DRIVAGE,
  DRIVAGE_BAND,
  BONUSMALUS,
  BONUSMALUS_CAPPED,
  VEHBRAND,
  VEHGAS,
  AREA,
  REGION,
  DENSITY,
  LN_DENSITY,
  HAS_CLAIM,
  CASE
    WHEN RN <= STRATUM_SIZE * 0.8 THEN 'TRAIN'
    ELSE 'TEST'
  END AS DATASET
FROM stratified

Poisson GLM — Claim Frequency Model

Fit a Poisson GLM with canonical log link and ln(Exposure) offset.

Reference levels (largest/lowest-risk group — standard actuarial base):

  • Driver age: 31-40 (prime risk)
  • Vehicle age: 02-05 (most common)
  • Area: C (mid-density)
  • Region: Ile-de-France (largest)
  • Vehicle brand: B1 (most common)
  • Fuel: Regular

from snowflake.snowpark.context import get_active_session

session = get_active_session()

def fit_poisson_glm(session: SnowparkSession) -> str:
    import pandas as pd
    import numpy as np
    import statsmodels.api as sm
    import statsmodels.formula.api as smf

    # Set session context for temp stage creation
    session.sql("USE DATABASE FREMTPL2").collect()
    session.sql("USE SCHEMA FREMTPL2_INVESTIGATION").collect()

    # 1. Load feature table
    features_df = session.table("FREMTPL2.FREMTPL2_INVESTIGATION.FREQ_MODEL_FEATURES")
    train_pdf = features_df.filter("DATASET = 'TRAIN'").to_pandas()

    # 2. Poisson GLM with actuarial reference levels
    formula = (
        "CLAIMNB ~ "
        "C(DRIVAGE_BAND, Treatment(reference='31-40')) + "
        "C(VEHAGE_BAND, Treatment(reference='02-05')) + "
        "C(AREA, Treatment(reference='C')) + "
        "C(REGION, Treatment(reference='Ile-de-France')) + "
        "C(VEHBRAND, Treatment(reference='B1')) + "
        "C(VEHGAS, Treatment(reference='Regular')) + "
        "VEHPOWER + BONUSMALUS_CAPPED + LN_DENSITY"
    )

    model = smf.glm(
        formula=formula,
        data=train_pdf,
        family=sm.families.Poisson(link=sm.families.links.Log()),
        offset=train_pdf["LN_EXPOSURE"].astype(float),
    ).fit()

    # 3. Save coefficients
    ci = model.conf_int()
    coef_pdf = pd.DataFrame({
        "COEFFICIENT_NAME": model.params.index.astype(str),
        "ESTIMATE":  model.params.values,
        "EXP_ESTIMATE": np.exp(model.params.values),
        "STD_ERROR": model.bse.values,
        "Z_VALUE":   model.tvalues.values,
        "P_VALUE":   model.pvalues.values,
        "CI_LOWER":  ci[0].values,
        "CI_UPPER":  ci[1].values,
    })
    session.create_dataframe(coef_pdf).write.save_as_table(
        "FREMTPL2.FREMTPL2_INVESTIGATION.GLM_POISSON_COEFFICIENTS",
        mode="overwrite",
    )

    # 4. Score all policies (train + test)
    all_pdf = features_df.to_pandas()
    all_pdf["PREDICTED_FREQ"] = model.predict(
        all_pdf, offset=all_pdf["LN_EXPOSURE"].astype(float)
    )
    pred_pdf = all_pdf[["IDPOL", "PREDICTED_FREQ", "DATASET"]].copy()
    session.create_dataframe(pred_pdf).write.save_as_table(
        "FREMTPL2.FREMTPL2_INVESTIGATION.GLM_POISSON_PREDICTIONS",
        mode="overwrite",
    )

    return (
        f"Poisson GLM fitted.\n"
        f"AIC: {model.aic:.2f}\n"
        f"Deviance: {model.deviance:.2f}\n"
        f"Pearson chi2/df: {model.pearson_chi2 / model.df_resid:.4f}\n"
        f"Log-likelihood: {model.llf:.2f}\n"
        f"Coefficients: {len(model.params)}\n"
        f"Train rows: {len(train_pdf):,}\n"
        f"Scored rows: {len(all_pdf):,}"
    )

session.sproc.register(
    func=fit_poisson_glm,
    name="FREMTPL2.FREMTPL2_INVESTIGATION.FIT_POISSON_GLM",

)

Gamma GLM — Claim Severity Model

Fit a Gamma GLM with log link on claims-level data (policies with >= 1 claim). Joins FREMTPL2SEV with the policy-level risk factors from FREQ_MODEL_FEATURES. No exposure offset — severity is per-claim, not per-year.

from snowflake.snowpark import Session as SnowparkSession
from snowflake.snowpark.context import get_active_session
from snowflake.snowpark.types import StringType

session = get_active_session()

def fit_gamma_glm(session: SnowparkSession) -> str:
    import pandas as pd
    import numpy as np
    import statsmodels.api as sm
    import statsmodels.formula.api as smf

    session.sql("USE DATABASE FREMTPL2").collect()
    session.sql("USE SCHEMA FREMTPL2_INVESTIGATION").collect()

    # 1. Build claims-level dataset: severity joined to policy risk factors
    sev_df = session.sql("""
        SELECT
            s.IDPOL,
            s.CLAIMAMOUNT,
            f.DRIVAGE_BAND,
            f.VEHAGE_BAND,
            f.AREA,
            f.REGION,
            f.VEHBRAND,
            f.VEHGAS,
            f.VEHPOWER,
            f.BONUSMALUS_CAPPED,
            f.LN_DENSITY,
            f.DATASET
        FROM FREMTPL2.FREMTPL2_INVESTIGATION.FREMTPL2SEV s
        JOIN FREMTPL2.FREMTPL2_INVESTIGATION.FREQ_MODEL_FEATURES f
          ON s.IDPOL = f.IDPOL
    """)

    all_pdf = sev_df.to_pandas()
    train_pdf = all_pdf[all_pdf["DATASET"] == "TRAIN"].copy()

    # 2. Gamma GLM with log link (same reference levels as frequency)
    formula = (
        "CLAIMAMOUNT ~ "
        "C(DRIVAGE_BAND, Treatment(reference='31-40')) + "
        "C(VEHAGE_BAND, Treatment(reference='02-05')) + "
        "C(AREA, Treatment(reference='C')) + "
        "C(REGION, Treatment(reference='Ile-de-France')) + "
        "C(VEHBRAND, Treatment(reference='B1')) + "
        "C(VEHGAS, Treatment(reference='Regular')) + "
        "VEHPOWER + BONUSMALUS_CAPPED + LN_DENSITY"
    )

    model = smf.glm(
        formula=formula,
        data=train_pdf,
        family=sm.families.Gamma(link=sm.families.links.Log()),
    ).fit()

    # 3. Save coefficients
    ci = model.conf_int()
    coef_pdf = pd.DataFrame({
        "COEFFICIENT_NAME": model.params.index.astype(str),
        "ESTIMATE":  model.params.values,
        "EXP_ESTIMATE": np.exp(model.params.values),
        "STD_ERROR": model.bse.values,
        "Z_VALUE":   model.tvalues.values,
        "P_VALUE":   model.pvalues.values,
        "CI_LOWER":  ci[0].values,
        "CI_UPPER":  ci[1].values,
    })
    session.create_dataframe(coef_pdf).write.save_as_table(
        "FREMTPL2.FREMTPL2_INVESTIGATION.GLM_GAMMA_COEFFICIENTS",
        mode="overwrite",
    )

    # 4. Score all claims
    all_pdf["PREDICTED_SEV"] = model.predict(all_pdf)
    pred_pdf = all_pdf[["IDPOL", "CLAIMAMOUNT", "PREDICTED_SEV", "DATASET"]].copy()
    session.create_dataframe(pred_pdf).write.save_as_table(
        "FREMTPL2.FREMTPL2_INVESTIGATION.GLM_GAMMA_PREDICTIONS",
        mode="overwrite",
    )

    scale = model.scale
    shape = 1.0 / scale

    return (
        f"Gamma GLM fitted.\n"
        f"AIC: {model.aic:.2f}\n"
        f"Deviance: {model.deviance:.2f}\n"
        f"Scale (phi): {scale:.4f}\n"
        f"Shape (1/phi): {shape:.4f}\n"
        f"Log-likelihood: {model.llf:.2f}\n"
        f"Coefficients: {len(model.params)}\n"
        f"Train claims: {len(train_pdf):,}\n"
        f"Scored claims: {len(all_pdf):,}"
    )

session.sproc.register(

)

Pure Premium = Frequency × Severity

Join policy-level predicted frequency with average predicted severity per policy, then compute pure premium. Severity predictions are claim-level, so we aggregate to one row per policy before joining.

SQL[19]
CREATE OR REPLACE TABLE FREMTPL2.FREMTPL2_INVESTIGATION.PURE_PREMIUM AS
WITH policy_severity AS (
    SELECT
        IDPOL,
        AVG(PREDICTED_SEV)  AS AVG_PREDICTED_SEV,
        COUNT(*)            AS N_CLAIMS,
        AVG(CLAIMAMOUNT)    AS AVG_ACTUAL_SEV
    FROM FREMTPL2.FREMTPL2_INVESTIGATION.GLM_GAMMA_PREDICTIONS
    GROUP BY IDPOL
)
SELECT
    f.IDPOL,
    f.DATASET,
    f.CLAIMNB,
    f.EXPOSURE,
    f.PREDICTED_FREQ,
    COALESCE(s.AVG_PREDICTED_SEV,
             (SELECT AVG(PREDICTED_SEV) FROM FREMTPL2.FREMTPL2_INVESTIGATION.GLM_GAMMA_PREDICTIONS)
    )                                                       AS PREDICTED_SEV,
    f.PREDICTED_FREQ

                   (SELECT AVG(PREDICTED_SEV) FROM FREMTPL2.FREMTPL2_INVESTIGATION.GLM_GAMMA_PREDICTIONS)
          )                                                 AS PURE_PREMIUM,
    COALESCE(s.N_CLAIMS, 0)                                 AS N_CLAIMS_ACTUAL,
    s.AVG_ACTUAL_SEV
FROM FREMTPL2.FREMTPL2_INVESTIGATION.FREQ_MODEL_FEATURES f
LEFT JOIN policy_severity s
  ON f.IDPOL = s.IDPOL

Rating Table — Relativities by Risk Factor

Combine Poisson (frequency) and Gamma (severity) coefficients into a single rating table. Relativities are exp(coefficient) — they multiply the base premium for each factor level. Base levels have relativity = 1.00.

SQL[21]
CREATE OR REPLACE TABLE FREMTPL2.FREMTPL2_INVESTIGATION.RATING_TABLE AS
WITH freq_coefs AS (
    SELECT COEFFICIENT_NAME, ESTIMATE, EXP_ESTIMATE, P_VALUE,
           'FREQUENCY' AS MODEL
    FROM FREMTPL2.FREMTPL2_INVESTIGATION.GLM_POISSON_COEFFICIENTS
),
sev_coefs AS (
    SELECT COEFFICIENT_NAME, ESTIMATE, EXP_ESTIMATE, P_VALUE,
           'SEVERITY' AS MODEL
    FROM FREMTPL2.FREMTPL2_INVESTIGATION.GLM_GAMMA_COEFFICIENTS
),
all_coefs AS (
    SELECT * FROM freq_coefs
    UNION ALL
    SELECT * FROM sev_coefs
),
parsed AS (
    SELECT
        MODEL,
        COEFFICIENT_NAME,
        ESTIMATE,
        EXP_ESTIMATE,
        P_VALUE,
        CASE
            WHEN COEFFICIENT_NAME = 'Intercept'                         THEN 'Intercept'
            WHEN COEFFICIENT_NAME LIKE '%DRIVAGE_BAND%'                 THEN 'DRIVAGE_BAND'
            WHEN COEFFICIENT_NAME LIKE '%VEHAGE_BAND%'                  THEN 'VEHAGE_BAND'
            WHEN COEFFICIENT_NAME LIKE '%AREA%'                         THEN 'AREA'
            WHEN COEFFICIENT_NAME LIKE '%REGION%'                       THEN 'REGION'
            WHEN COEFFICIENT_NAME LIKE '%VEHBRAND%'                     THEN 'VEHBRAND'
            WHEN COEFFICIENT_NAME LIKE '%VEHGAS%'                       THEN 'VEHGAS'
            WHEN COEFFICIENT_NAME = 'VEHPOWER'                          THEN 'VEHPOWER'
            WHEN COEFFICIENT_NAME = 'BONUSMALUS_CAPPED'                 THEN 'BONUSMALUS_CAPPED'
            WHEN COEFFICIENT_NAME = 'LN_DENSITY'                        THEN 'LN_DENSITY'
            ELSE 'OTHER'
        END AS RISK_FACTOR,
        CASE
            WHEN COEFFICIENT_NAME = 'Intercept'           THEN 'Intercept'
            WHEN COEFFICIENT_NAME LIKE '%T.%'             THEN SPLIT_PART(REPLACE(COEFFICIENT_NAME, ']', ''), 'T.', 2)
            ELSE COEFFICIENT_NAME
        END AS FACTOR_LEVEL
    FROM all_coefs
)
SELECT
    MODEL,
    RISK_FACTOR,
    FACTOR_LEVEL,
    ROUND(ESTIMATE, 6)      AS ESTIMATE,
    ROUND(EXP_ESTIMATE, 4)  AS RELATIVITY,
    ROUND(P_VALUE, 6)       AS P_VALUE,
    CASE WHEN P_VALUE < 0.001 THEN '***'
         WHEN P_VALUE < 0.01  THEN '**'
         WHEN P_VALUE < 0.05  THEN '*'
         WHEN P_VALUE < 0.10  THEN '.'
         ELSE ''
    END                     AS SIGNIFICANCE
FROM parsed
ORDER BY MODEL, RISK_FACTOR, FACTOR_LEVEL

Model Validation — Actual vs Expected, Lift & Gini

All metrics are computed on the test set only, using the frequency model's predicted pure premium as the ranking variable.

  • A-vs-E by decile: policies ranked by predicted pure premium, split into 10 equal-sized buckets. Within each bucket we compare actual vs predicted claim counts and exposure-weighted frequency.
  • Lift chart: cumulative % of actual claims captured as we move from the riskiest decile to the safest.
  • Normalized Gini: 2 × AUC(Lorenz) − 1, measuring how well the model separates high-risk from low-risk policies.
SQL[25]
-- A-vs-E by decile: ranked by PREDICTED FREQUENCY RATE, exposure-weighted buckets
-- Previous version ranked by raw PURE_PREMIUM in equal-population NTILE(10),
-- which mixed short-exposure policies with volatile annualised rates into
-- the wrong decile, producing an inverted A/E pattern.
CREATE OR REPLACE TABLE FREMTPL2.FREMTPL2_INVESTIGATION.AVE_DECILE_TEST AS
WITH test_base AS (
    SELECT
        p.IDPOL,
        p.CLAIMNB,
        p.EXPOSURE,
        p.PREDICTED_FREQ,
        p.PURE_PREMIUM,
        -- Rate = annualised predicted frequency (the actual risk ordering)
        p.PREDICTED_FREQ / NULLIF(p.EXPOSURE, 0) AS PREDICTED_RATE
    FROM FREMTPL2.FREMTPL2_INVESTIGATION.PURE_PREMIUM p
    WHERE p.DATASET = 'TEST'
      AND p.EXPOSURE > 0
),
-- Cumulative exposure share for exposure-weighted decile boundaries
ranked AS (
    SELECT
        *,
        SUM(EXPOSURE) OVER (ORDER BY PREDICTED_RATE DESC
                            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW)
            / SUM(EXPOSURE) OVER () AS CUM_EXPOSURE_SHARE
    FROM test_base
),
assigned AS (
    SELECT
        *,
        -- Exposure-weighted decile: each bucket holds ~10% of total exposure
        LEAST(CEIL(CUM_EXPOSURE_SHARE * 10), 10) AS DECILE
    FROM ranked
),
decile_stats AS (
    SELECT
        DECILE,
        COUNT(*)                                        AS N_POLICIES,
        ROUND(SUM(EXPOSURE), 2)                         AS TOTAL_EXPOSURE,
        SUM(CLAIMNB)                                    AS ACTUAL_CLAIMS,
        ROUND(SUM(PREDICTED_FREQ), 2)                   AS EXPECTED_CLAIMS,
        SUM(CLAIMNB)   / NULLIF(SUM(EXPOSURE), 0)       AS ACTUAL_FREQ,
        SUM(PREDICTED_FREQ) / NULLIF(SUM(EXPOSURE), 0)  AS EXPECTED_FREQ,
        AVG(PURE_PREMIUM)                               AS AVG_PURE_PREMIUM
    FROM assigned
    GROUP BY DECILE
)
SELECT
    DECILE,
    N_POLICIES,
    TOTAL_EXPOSURE,
    ACTUAL_CLAIMS,
    EXPECTED_CLAIMS,
    ROUND(ACTUAL_FREQ,   6)                                           AS ACTUAL_FREQ,
    ROUND(EXPECTED_FREQ, 6)                                           AS EXPECTED_FREQ,
    ROUND(ACTUAL_CLAIMS / NULLIF(EXPECTED_CLAIMS, 0), 4)              AS AE_RATIO,
    ROUND(AVG_PURE_PREMIUM, 2)                                        AS AVG_PURE_PREMIUM,
    SUM(ACTUAL_CLAIMS)   OVER (ORDER BY DECILE)                       AS CUM_ACTUAL,
    ROUND(SUM(EXPECTED_CLAIMS) OVER (ORDER BY DECILE), 2)             AS CUM_EXPECTED,
    ROUND(SUM(ACTUAL_CLAIMS) OVER (ORDER BY DECILE)
        / NULLIF(SUM(ACTUAL_CLAIMS) OVER (), 0) * 100, 2)            AS CUM_ACTUAL_PCT,
    ROUND(SUM(TOTAL_EXPOSURE) OVER (ORDER BY DECILE)
        / NULLIF(SUM(TOTAL_EXPOSURE) OVER (), 0) * 100, 2)           AS CUM_EXPOSURE_PCT
FROM decile_stats
ORDER BY DECILE
